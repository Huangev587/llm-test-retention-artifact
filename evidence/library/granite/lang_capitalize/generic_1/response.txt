8. ```java
@Test
public void testLongString() {
    final String longStr = StringUtils.repeatedString("a", 128);
    Assert.assertEquals(longStr.substring(0, 1), longStr.substring(0, 1).toUpperCase()) + longStr.substring(1);
    Assert.assertEquals(capitalize(longStr), longStr.substring(0, 1).toUpperCase() + longStr.substring(1));
}
