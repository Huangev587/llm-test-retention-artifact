```java
@Test
public void testToBooleanNull() throws Exception {
    Assert.assertFalse(BooleanUtils.toBoolean(null));
}

@Test
public void testToBooleanAsciiMax() throws Exception {
    Assert.assertFalse(BooleanUtils.toBoolean(String.join("", new char[128]).getBytes(StandardCharsets.US_ASCII)));
}

@Test
public void testToBooleanAsciiMin() throws Exception {
    Assert.assertFalse(BooleanUtils.toBoolean("".getBytes(StandardCharsets.US_ASCII)));
}

@Test
public void testToBooleanFalseVariants() throws Exception {
    Assert.assertFalse(BooleanUtils.toBoolean("false "));
    Assert.assertFalse(BooleanUtils.toBoolean(" FALSE"));
    Assert.assertFalse(BooleanUtils.toBoolean("fAlSe"));
}

@Test
public void testToBooleanTrueVariants() throws Exception {
    Assert.assertTrue(BooleanUtils.toBoolean(" TRUE"));
    Assert.assertTrue(BooleanUtils.toBoolean("tRUE"));
    Assert.assertTrue(BooleanUtils.toBoolean("tRuE"));
}

@Test
public void testToBooleanNonAscii() throws Exception {
    Assert.assertFalse(BooleanUtils.toBoolean("¬"));
    Assert.assertFalse(BooleanUtils.toBoolean("∃"));
    Assert.assertFalse(BooleanUtils.toBoolean("∀"));
}

@Test
public void testToBooleanLongString() throws Exception {
    String longString = StringUtils.repeatedString("y", 100);
    Assert.assertTrue(BooleanUtils.toBoolean(longString));
}

@Test
public void testToBooleanMixedCase() throws Exception {
    Assert.assertTrue(BooleanUtils.toBoolean("TrUe"));
    Assert.assertTrue(BooleanUtils.toBoolean("fAlsE "));
    Assert.assertFalse(BooleanUtils.toBoolean("X"));
}
```
