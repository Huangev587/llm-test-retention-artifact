```java
@Test
public void testIndexOfNullArray() {
    assertArrayIndexOf(-1, (int[]) null);
}

@Test
public void testIndexOfEmptyArray() {
    int[] emptyArray = {};
    assertArrayIndexOf(-1, emptyArray);
}

@Test
public void testIndexOfSingleElementArray() {
    int[] singleElementArray = {5};
    assertArrayIndexOf(0, singleElementArray);
}

@Test
public void testIndexOfMultipleElementsArray() {
    int[] multipleElementsArray = {1, 2, 3, 4, 5};
    assertArrayIndexOf(2, multipleElementsArray);
}

@Test
public void testIndexOfValueNotInArray() {
    int[] array = {1, 2, 3, 4, 5};
    assertArrayIndexOf(6, array);
}

@Test
public void testIndexOfValueAtArrayEnd() {
    int[] array = {1, 2, 3, 4, 5};
    assertArrayIndexOf(5, array);
}

@Test
public void testIndexOfValueAtArrayStart() {
    int[] array = {5, 1, 2, 3, 4};
    assertArrayIndexOf(5, array);
}

@Test
public void testIndexOfValueRepeatedMultipleTimes() {
    int[] array = {5, 5, 5, 5, 5};
    assertArrayIndexOf(0, array);
}

private void assertArrayIndexOf(int expectedIndex, int[] array) {
    int actualIndex = ArrayUtils.indexOf(array, 5);
    assertEquals(expectedIndex, actualIndex);
}
```
