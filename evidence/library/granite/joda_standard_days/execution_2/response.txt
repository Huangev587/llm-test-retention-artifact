1. ```java
@Test
public void testStandardDaysZero() {
    Duration result = Duration.standardDays(0);
    Assert.assertEquals("ZERO", result.toString(), Duration.ZERO.toString());
}
```

2. ```java
@Test
public void testStandardDaysPositive() {
    long days = 1;
    Duration expected = Duration.standardDays(days);
    Duration result = Duration.standardDays(days);
    Assert.assertEquals("Expected: " + expected.getMillis() + ", Got: " + result.getMillis(), days * DateTimeConstants.MILLIS_PER_DAY, result.getMillis());
}
```

7. ```java
@Test
public void testStandardDaysWithHours() {
    long days = 1;
    long hours = 24;
    Duration expected = Duration.standardDays(days).plusHours(hours);
    Duration result = Duration.standardDays(days).plus(Duration.standardHours(hours));
    Assert.assertEquals("Expected: " + expected.getMillis() + ", Got: " + result.getMillis(), expected.getMillis(), result.getMillis());
}
