1. @Test public void testNullInput() {
    Assert.assertNull(StringUtils.leftPad(null, 3, 'z'));
}

2. @Test public void testEmptyStringInput() {
    Assert.assertEquals("zzzz", StringUtils.leftPad("", 3, 'z'));
}

3. @Test public void testShortStringInput() {
    Assert.assertEquals("bat", StringUtils.leftPad("bat", 3, 'z'));
}

4. @Test public void testLongStringInput() {
    Assert.assertEquals("zzbat", StringUtils.leftPad("bat", 5, 'z'));
}

5. @Test public void testStringAlreadyLargeEnough() {
    Assert.assertEquals("bat", StringUtils.leftPad("bat", 1, 'z'));
}

6. @Test public void testNegativeSizeInput() {
    Assert.assertEquals("bat", StringUtils.leftPad("bat", -1, 'z'));
}

7. @Test public void testLargePadding() {
    Assert.assertEquals("zzzzzzzzzzzzzzzzz", StringUtils.leftPad("bat", 128, 'z'));
}

8. @Test public void testLargeNegativePadding() {
    Assert.assertEquals("bat", StringUtils.leftPad("bat", -128, 'z'));
