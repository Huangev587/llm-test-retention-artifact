1. ```java
@Test
public void testWithDayOfMonth() {
    LocalDate date = LocalDate.of(1900, 1, 1);
    LocalDate result = date.withDayOfMonth(15);
    assertArrayEquals(new int[]{1900, 1, 15}, result.toInstant().atZone(ZoneOffset.UTC).toLocalDate().getValues());
}
```
