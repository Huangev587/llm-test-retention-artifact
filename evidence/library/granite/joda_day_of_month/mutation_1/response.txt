1. @Test public void testWithDayOfMonth() throws Exception {
   LocalDate date = LocalDate.of(1900, 1, 1);
   LocalDate result = date.withDayOfMonth(15);
   assertArrayEquals(new int[]{1900, 1, 15}, result.toInstant().toEpochMilli() / 10000000000L >> 32);
}
