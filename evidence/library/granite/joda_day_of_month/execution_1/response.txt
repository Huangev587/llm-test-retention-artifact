1. ```java
@Test
public void testWithDayOfMonth() {
    LocalDate original = LocalDate.of(1900, 1, 1);
    LocalDate expected = LocalDate.of(1900, 1, 15);
    assertArrayEquals(expected.toString().split("-"), original.withDayOfMonth(15).toString().split("-"));
}
