public void testWithDayOfMonthValidInput() throws Exception {
    LocalDate date = LocalDate.of(1950, 2, 28);

    assertArrayEquals(new int[]{1950, 3, 29}, result.toInstant().atZone(ZoneId.systemDefault()).toLocalDate().getValues());
}
