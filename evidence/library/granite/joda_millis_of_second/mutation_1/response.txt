1. @Test public void testWithMillisOfSecond() throws Exception {
   LocalTime time = LocalTime.of(12, 30, 45, 123);
   LocalTime result = time.withMillisOfSecond(456);
   Assert.assertEquals(12, result.getHour());
   Assert.assertEquals(30, result.getMinuteOfHour());
   Assert.assertEquals(45, result.getSecondOfMinute());
   Assert.assertEquals(456, result.getMillisOfSecond());
}

2. @Test public void testWithMillisOfSecondOverflow() throws Exception {
   LocalTime time = LocalTime.of(12, 30, 59, 999);
   try {
       time.withMillisOfSecond(100);
       Assert.fail("Expected IllegalArgumentException");
   } catch (IllegalArgumentException e) {
       Assert.assertTrue(true);
   }
}
