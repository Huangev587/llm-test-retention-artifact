1. ```java
@Test
public void testWithMillisOfSecond() {
    LocalTime time = LocalTime.of(12, 34, 56, 789);
    LocalTime result = time.withMillisOfSecond(123);
    assertEquals(12, result.getHour());
    assertEquals(34, result.getMinuteOfHour());
    assertEquals(56, result.getSecondOfMinute());
    assertEquals(123, result.getMillisOfSecond());
}
```
