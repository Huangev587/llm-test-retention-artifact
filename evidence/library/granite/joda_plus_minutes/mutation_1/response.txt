@Test
public void testPlusMinutesZero() {
    LocalTime time = LocalTime.of(12, 30, 0);
    LocalTime result = time.plusMinutes(0);
    assertEquals(12, result.getHour());
    assertEquals(30, result.getMinute());
    assertEquals(0, result.getSecond());
    assertEquals(0, result.getMillis());
}
