```java
@Test
public void testParseValidDuration() throws Exception {
    Duration duration = Duration.parse("P1DT2H3M4S");
    assertEquals("P1DT2H3M4S", duration.toString());
    assertEquals(1, duration.getDays());
    assertEquals(2, duration.getHours());
    assertEquals(3, duration.getMinutes());
    assertEquals(4, duration.getSeconds());
}
```
