1. ```java
@Test
public void testParsePositiveDuration() {
    String str = "PT12345.678S";
    Duration duration = Duration.parse(str);
    assertEquals("PT12345.678S", duration.toString());
    assertEquals(1234567800000L, duration.getMillis());
}
```
