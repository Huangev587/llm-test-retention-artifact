    LocalDate result = original.plusDays(0);
    assertEquals(original, result);
}

@Test
public void testPlusDaysPositive() {
    LocalDate original = new LocalDate(1900, 1, 1);
    LocalDate result = original.plusDays(5);
    assertEquals(original.plusDays(5), result);
}

public void testPlusDaysNegative() {

}
