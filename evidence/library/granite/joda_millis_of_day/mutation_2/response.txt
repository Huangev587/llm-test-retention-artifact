1. ```java
@Test
public void testGetMillisOfDay() throws Exception {
    LocalTime localTime = new LocalTime(12, 34, 56, 789);
    int expected = localTime.getHour() * 3600000 + localTime.getMinute() * 60000 + localTime.getSecond() * 1000 + localTime.getMillis();
    int actual = localTime.getMillisOfDay();
    Assert.assertEquals(expected, actual);
}
```
