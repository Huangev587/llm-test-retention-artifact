@Test
public void testPlusMonthsZero() {
    LocalDate original = new LocalDate(1950, 1, 15);
    LocalDate result = original.plusMonths(0);
    assertEquals(original, result);
}

@Test
public void testPlusMonthsPositive() {
    LocalDate original = new LocalDate(1950, 1, 15);
    LocalDate result = original.plusMonths(3);
    assertEquals(new LocalDate(1950, 4, 15), result);
}

@Test
public void testPlusMonthsNegative() {
    LocalDate original = new LocalDate(1950, 1, 15);
    LocalDate result = original.plusMonths(-3);
    assertEquals(new LocalDate(1949, 10, 15), result);
}

@Test
public void testPlusMonthsEndOfMonth() {
    LocalDate original = new LocalDate(1950, 1, 31);
    LocalDate result = original.plusMonths(1);
    assertEquals(new LocalDate(1950, 2, 28), result);
}

@Test
public void testPlusMonthsEndOfMonthLeapYear() {
    LocalDate original = new LocalDate(1952, 12, 31);
    LocalDate result = original.plusMonths(1);
    assertEquals(new LocalDate(1953, 2, 28), result);
}

@Test
public void testPlusMonthsStartOfMonth() {
    LocalDate original = new LocalDate(1950, 2, 1);
    LocalDate result = original.plusMonths(1);
    assertEquals(new LocalDate(1950, 3, 1), result);
}

@Test
public void testPlusMonthsEndOfMonthClamping() {
    LocalDate original = new LocalDate(1950, 1, 30);
    LocalDate result = original.plusMonths(1);
    assertEquals(new LocalDate(1950, 2, 28), result);
}

@Test
public void testPlusMonthsEndOfMonthClampingLeapYear() {
    LocalDate original = new LocalDate(1952, 12, 30);
    LocalDate result = original.plusMonths(1);
    assertEquals(new LocalDate(1953, 2, 28), result);
}
