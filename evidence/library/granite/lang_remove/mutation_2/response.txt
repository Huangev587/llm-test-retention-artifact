@Test
public void testRemoveLastElementArray() {
    int[] input = {2, 6};
    int index = 1;
    int[] expected = {2};
    int[] result = ArrayUtils.remove(input, index);
    Assert.assertArrayEquals(expected, result);
}

@Test
public void testRemoveElementFromLongArray() {
    int[] input = {1, 2, 3, 4, 5};
    int index = 2;
    int[] expected = {1, 2, 4, 5};
    int[] result = ArrayUtils.remove(input, index);
    Assert.assertArrayEquals(expected, result);
}

@Test
public void testRemoveElementFromLargeArray() {
    int[] input = new int[64];
    for (int i = 0; i < 64; i++) {
        input[i] = i + 1;
    }
    int index = 32;
    int[] expected = new int[64];
    System.arraycopy(input, 0, expected, 0, index);
    System.arraycopy(input, index + 1, expected, index, 63 - index);
    int[] result = ArrayUtils.remove(input, index);
    Assert.assertArrayEquals(expected, result);
}

@Test
