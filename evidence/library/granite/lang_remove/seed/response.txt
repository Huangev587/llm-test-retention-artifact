1. @Test
public void testRemoveFromEmptyArray() {
    int[] array = {};
    int index = 0;
    int[] expected = {};
    assertArrayEquals(expected, remove(array, index));
}

2. @Test
public void testRemoveFromSingleElementArray() {
    int[] array = {1};
    int index = 0;
    int[] expected = {};
    assertArrayEquals(expected, remove(array, index));
}

3. @Test
public void testRemoveFromMultipleElementArray() {
    int[] array = {2, 6};
    int index = 0;
    int[] expected = {6};
    assertArrayEquals(expected, remove(array, index));
}

4. @Test
public void testRemoveFromMultipleElementArrayMiddle() {
    int[] array = {2, 6};
    int index = 1;
    int[] expected = {2};
    assertArrayEquals(expected, remove(array, index));
}

5. @Test
public void testRemoveFromMultipleElementArrayLast() {
    int[] array = {2, 6, 3};
    int index = 2;
    int[] expected = {2, 3};
    assertArrayEquals(expected, remove(array, index));
}

6. @Test
public void testRemoveFromLargeArray() {
    int[] array = new int[64];
    Arrays.fill(array, 1);
    int index = 32;
    int[] expected = new int[63];
    Arrays.fill(expected, 1);
    assertArrayEquals(expected, remove(array, index));
}

7. @Test
public void testRemoveWithNegativeIndex() {
    int[] array = {1, 2, 3};
    int index = -1;
    int[] expected = {1, 2, 3};
    assertArrayEquals(expected, remove(array, index));
}

8. @Test(expected = IndexOutOfBoundsException.class)
public void testRemoveWithOutOfRangeIndex() {
    int[] array = {1, 2, 3};
    int index = 3;
    remove(array, index);
}
