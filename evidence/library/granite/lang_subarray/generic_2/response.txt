2. ```java
@Test
public void testSubarrayEmptyArray() {

    assertArrayEquals(EMPTY_INT_ARRAY, result);
}
```

3. ```java
@Test
public void testSubarraySingleElement() {
    int[] values = {1};
    int[] result = subarray(values, 0, 1);
    assertArrayEquals(new int[]{1}, result);
}
```

4. ```java
@Test
public void testSubarrayFullArray() {
    int[] values = new int[10];
    for (int i = 0; i < values.length; i++) {
        values[i] = i;
    }
    int[] result = subarray(values, 0, 11);
    assertArrayEquals(values, result);
}
```

5. ```java
