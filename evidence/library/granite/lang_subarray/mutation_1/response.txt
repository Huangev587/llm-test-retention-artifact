@Test
public void testSubarrayNullInput() {
    int[] result = subarray(null, 0, 10);
    assertNull(result);
}

@Test
public void testSubarrayEmptyArray() {
    int[] array = {};
    int[] result = subarray(array, 0, 10);
    assertArrayEquals(EMPTY_INT_ARRAY, result);
}

@Test
public void testSubarraySingleElement() {
    int[] array = {1};
    int[] result = subarray(array, 0, 1);
    assertArrayEquals(new int[]{1}, result);
}

@Test
public void testSubarrayFullArray() {
    int[] array = new int[10];
    for (int i = 0; i < array.length; i++) {
        array[i] = i + 1;
    }
    int[] result = subarray(array, 0, 10);
    assertArrayEquals(array, result);
}

@Test
public void testSubarrayFromMiddle() {
    int[] array = {1, 2, 3, 4, 5};
    int[] result = subarray(array, 1, 4);
    assertArrayEquals(new int[]{2, 3, 4}, result);
}

@Test
public void testSubarrayToEnd() {
    int[] array = {1, 2, 3, 4, 5};
    int[] result = subarray(array, 2, 6);
    assertArrayEquals(new int[]{3, 4, 5}, result);
}

@Test
public void testSubarrayWithNegativeIndex() {
    int[] array = {1, 2, 3, 4, 5};
    int[] result = subarray(array, -1, 3);
    assertArrayEquals(new int[]{4, 5}, result);
}

@Test
public void testSubarrayOversizedEndIndex() {
    int[] array = {1, 2, 3, 4, 5};
    int[] result = subarray(array, 0, 10);
    assertArrayEquals(array, result);
}
