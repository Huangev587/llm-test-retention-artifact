@Test
public void testPlusDaysZero() throws Exception {
    LocalDate base = new LocalDate(2020, 1, 1);
    LocalDate result = base.plusDays(0);
    assertEquals("2020-01-01", result.toString());
}

@Test
public void testPlusDaysPositive() throws Exception {
    LocalDate base = new LocalDate(2020, 1, 1);
    LocalDate result = base.plusDays(5);
    assertEquals("2020-01-06", result.toString());
}

@Test
public void testPlusDaysNegative() throws Exception {
    LocalDate base = new LocalDate(2020, 1, 1);
    LocalDate result = base.plusDays(-5);
    assertEquals("1999-12-27", result.toString());
}

@Test
public void testPlusDaysMaxPositive() throws Exception {
    LocalDate base = new LocalDate(2100, 12, 31);
    LocalDate result = base.plusDays(36600);
    assertEquals("2460-01-01", result.toString());
}

@Test
public void testPlusDaysMinNegative() throws Exception {
    LocalDate base = new LocalDate(1900, 1, 1);
    LocalDate result = base.plusDays(-36600);
    assertEquals("-1160-12-31", result.toString());
}

@Test
public void testPlusDaysBoundaryPositive() throws Exception {
    LocalDate base = new LocalDate(2000, 2, 29);
    LocalDate result = base.plusDays(1);
    assertEquals("2000-03-01", result.toString());
}

@Test
public void testPlusDaysBoundaryNegative() throws Exception {
    LocalDate base = new LocalDate(2000, 3, 1);
    LocalDate result = base.plusDays(-1);
    assertEquals("2000-02-29", result.toString());
}

@Test
public void testPlusDaysLeapYear() throws Exception {
    LocalDate base = new LocalDate(2020, 2, 29);
    LocalDate result = base.plusDays(366);

}
