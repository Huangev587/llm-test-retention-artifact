@Test
public void testWithDayOfMonthValidDay() throws Exception {
    LocalDate original = new LocalDate(2023, 10, 5);
    LocalDate result = original.withDayOfMonth(10);
    assertEquals("2023-10-10", result.toString());
}

@Test
public void testWithDayOfMonthFirstDayOfMonth() throws Exception {
    LocalDate original = new LocalDate(2023, 10, 5);
    LocalDate result = original.withDayOfMonth(1);
    assertEquals("2023-10-01", result.toString());
}

@Test
public void testWithDayOfMonthLastDayOfMonth() throws Exception {
    LocalDate original = new LocalDate(2023, 10, 5);
    LocalDate result = original.withDayOfMonth(31);
    assertEquals("2023-10-31", result.toString());
}

@Test(expected = IllegalArgumentException.class)
public void testWithDayOfMonthInvalidDay() throws Exception {
    LocalDate original = new LocalDate(2023, 10, 5);
    original.withDayOfMonth(32);
}

@Test
public void testWithDayOfMonthLeapYear() throws Exception {
    LocalDate original = new LocalDate(2020, 2, 29);
    LocalDate result = original.withDayOfMonth(29);
    assertEquals("2020-02-29", result.toString());
}

@Test
public void testWithDayOfMonthNonLeapYear() throws Exception {
    LocalDate original = new LocalDate(2021, 2, 29);
    LocalDate result = original.withDayOfMonth(28);
    assertEquals("2021-02-28", result.toString());
}

@Test
public void testWithDayOfMonthSameMonthDifferentYear() throws Exception {
    LocalDate original = new LocalDate(2023, 10, 5);
    LocalDate result = original.withDayOfMonth(5).withYear(2024);
    assertEquals("2024-10-05", result.toString());
}

@Test
public void testWithDayOfMonthSameDayDifferentMonth() throws Exception {
    LocalDate original = new LocalDate(2023, 10, 5);
    LocalDate result = original.withDayOfMonth(5).withMonthOfYear(11);
    assertEquals("2023-11-05", result.toString());
}
