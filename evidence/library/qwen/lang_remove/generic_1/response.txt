@Test(expected = IndexOutOfBoundsException.class)
public void testRemoveNegativeIndex() throws Exception {
    ArrayUtils.remove(new int[]{1, 2, 3}, -1);
}

@Test(expected = IndexOutOfBoundsException.class)
public void testRemoveTooLargeIndex() throws Exception {
    ArrayUtils.remove(new int[]{1, 2, 3}, 3);
}

@Test
public void testRemoveFirstElement() throws Exception {
    assertArrayEquals(new int[]{2, 3}, ArrayUtils.remove(new int[]{1, 2, 3}, 0));
}

@Test
public void testRemoveLastElement() throws Exception {
    assertArrayEquals(new int[]{1, 2}, ArrayUtils.remove(new int[]{1, 2, 3}, 2));
}

@Test
public void testRemoveMiddleElement() throws Exception {
    assertArrayEquals(new int[]{1, 3}, ArrayUtils.remove(new int[]{1, 2, 3}, 1));
}

@Test
public void testRemoveSingleElementArray() throws Exception {
    assertArrayEquals(new int[]{}, ArrayUtils.remove(new int[]{1}, 0));
}

@Test
public void testRemoveFromEndOfLongArray() throws Exception {
    int[] original = new int[64];
    for (int i = 0; i < 64; i++) {
        original[i] = i;
    }
    int[] result = ArrayUtils.remove(original, 63);
    assertEquals(63, result.length);
    for (int i = 0; i < 63; i++) {
        assertEquals(i, result[i]);
    }
}

@Test
public void testRemoveFromBeginningOfLongArray() throws Exception {
    int[] original = new int[64];
    for (int i = 0; i < 64; i++) {
        original[i] = i;
    }
    int[] result = ArrayUtils.remove(original, 0);
    assertEquals(63, result.length);
    for (int i = 0; i < 63; i++) {
        assertEquals(i + 1, result[i]);
    }
}
