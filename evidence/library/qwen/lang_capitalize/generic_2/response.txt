@Test
public void testLeadingWhitespace() throws Exception {
    assertEquals(" Hello", StringUtils.capitalize(" hello"));
}

@Test
public void testMixedCase() throws Exception {
    assertEquals("Hello World", StringUtils.capitalize("hello world"));
}

@Test
public void testAlreadyCapitalized() throws Exception {
    assertEquals("Hello", StringUtils.capitalize("Hello"));
}

@Test
public void testAllUppercase() throws Exception {
    assertEquals("HELLO", StringUtils.capitalize("HELLO"));
}

@Test
public void testSpecialCharacters() throws Exception {
    assertEquals("!@#$%", StringUtils.capitalize("!@#$%"));
}

@Test
public void testLongString() throws Exception {
    String longStr = "abcdefghijklmnopqrstuvwxyz";
    assertEquals("Abcdefghijklmnopqrstuvwxyz", StringUtils.capitalize(longStr));
}
