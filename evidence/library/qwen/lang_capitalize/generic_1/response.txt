@Test
public void testMultipleCharsLowercase() throws Exception {
    assertEquals("Hello", StringUtils.capitalize("hello"));
}

@Test
public void testMultipleCharsMixedCase() throws Exception {
    assertEquals("Hello World", StringUtils.capitalize("hello world"));
}

@Test
public void testLeadingWhitespace() throws Exception {
    assertEquals(" Hello", StringUtils.capitalize(" hello"));
}

@Test
public void testTrailingWhitespace() throws Exception {
    assertEquals("Hello ", StringUtils.capitalize("hello "));
}

@Test
public void testInternalWhitespace() throws Exception {
    assertEquals("Hello World", StringUtils.capitalize("hello world"));
}

@Test
public void testSpecialCharacters() throws Exception {
    assertEquals("!@#$%^&*()", StringUtils.capitalize("!@#$%^&*()"));
}

@Test
public void testUnicodeSurrogatePair() throws Exception {
    assertEquals("\uD83D\uDE00", StringUtils.capitalize("\uD83D\uDE00"));
}
