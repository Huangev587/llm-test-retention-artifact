```java
@Test
public void testNullArray() throws Exception {
    assertEquals(-1, ArrayUtils.indexOf(null, 5));
}

@Test
public void testSingleElementMatch() throws Exception {
    assertEquals(0, ArrayUtils.indexOf(new int[]{5}, 5));
}

@Test
public void testSingleElementNoMatch() throws Exception {
    assertEquals(-1, ArrayUtils.indexOf(new int[]{5}, 4));
}

@Test
public void testLargeArrayNoMatch() throws Exception {
    int[] largeArray = new int[64];
    Arrays.fill(largeArray, 1000000);
    assertEquals(-1, ArrayUtils.indexOf(largeArray, 5));
}

@Test
public void testLargeArrayMatch() throws Exception {
    int[] largeArray = new int[64];
    Arrays.fill(largeArray, 1000000);
    largeArray[32] = 5;
    assertEquals(32, ArrayUtils.indexOf(largeArray, 5));
}

@Test
public void testNegativeValues() throws Exception {
    assertEquals(1, ArrayUtils.indexOf(new int[]{-1, -2, -3}, -2));
}

@Test
public void testPositiveValues() throws Exception {
    assertEquals(1, ArrayUtils.indexOf(new int[]{1, 2, 3}, 2));
}
```
