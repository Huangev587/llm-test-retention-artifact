@Test
public void testPlusMinutesWrapHourForward() throws Exception {
    LocalTime base = new LocalTime(23, 59, 59, 999);
    LocalTime result = base.plusMinutes(1);
    assertEquals("00:00:00.000", result.toString());
}

@Test
public void testPlusMinutesWrapHourBackward() throws Exception {
    LocalTime base = new LocalTime(0, 0, 0, 0);
    LocalTime result = base.plusMinutes(-1);
    assertEquals("23:59:59.999", result.toString());
}

@Test
public void testPlusMinutesLargePositive() throws Exception {
    LocalTime base = new LocalTime(12, 30, 45, 500);
    LocalTime result = base.plusMinutes(14400); // 24 hours
    assertEquals("12:30:45.500", result.toString());
}

@Test
public void testPlusMinutesLargeNegative() throws Exception {
    LocalTime base = new LocalTime(12, 30, 45, 500);
    LocalTime result = base.plusMinutes(-14400); // 24 hours
    assertEquals("12:30:45.500", result.toString());
}

@Test
public void testPlusMinutesCrossMidnightForward() throws Exception {
    LocalTime base = new LocalTime(23, 0, 0, 0);
    LocalTime result = base.plusMinutes(60);
    assertEquals("00:01:00.000", result.toString());
}

@Test
public void testPlusMinutesCrossMidnightBackward() throws Exception {
    LocalTime base = new LocalTime(0, 0, 0, 0);
    LocalTime result = base.plusMinutes(-60);
    assertEquals("23:59:00.000", result.toString());
}

@Test
public void testPlusMinutesBoundaryCondition() throws Exception {
    LocalTime base = new LocalTime(0, 0, 0, 0);
    LocalTime result = base.plusMinutes(1439); // Just before midnight
    assertEquals("23:59:00.000", result.toString());
}
