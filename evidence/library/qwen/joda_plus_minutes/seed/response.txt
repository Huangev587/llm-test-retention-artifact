@Test
public void testPlusMinutesZero() throws Exception {
    LocalTime base = new LocalTime(12, 30, 45, 500);
    LocalTime result = base.plusMinutes(0);
    assertEquals("12:30:45.500", result.toString());
}

@Test
public void testPlusMinutesPositive() throws Exception {
    LocalTime base = new LocalTime(12, 30, 45, 500);
    LocalTime result = base.plusMinutes(30);
    assertEquals("13:00:45.500", result.toString());
}

@Test
public void testPlusMinutesNegative() throws Exception {
    LocalTime base = new LocalTime(12, 30, 45, 500);
    LocalTime result = base.plusMinutes(-30);
    assertEquals("12:00:45.500", result.toString());
}

public void testPlusMinutesWrapHour() throws Exception {

}
