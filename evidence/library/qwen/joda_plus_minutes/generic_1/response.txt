@Test
public void testPlusMinutesWrapAroundMidnight() throws Exception {
    LocalTime base = new LocalTime(23, 59, 59, 999);
    LocalTime result = base.plusMinutes(1);
    assertEquals("00:00:00.000", result.toString());
}

@Test
public void testPlusMinutesWrapAroundNoSeconds() throws Exception {
    LocalTime base = new LocalTime(23, 59, 0, 0);
    LocalTime result = base.plusMinutes(1);
    assertEquals("00:00:00.000", result.toString());
}

@Test
public void testPlusMinutesWrapAroundWithSeconds() throws Exception {
    LocalTime base = new LocalTime(23, 59, 30, 500);
    LocalTime result = base.plusMinutes(1);
    assertEquals("00:00:30.500", result.toString());
}

@Test
public void testPlusMinutesLargePositive() throws Exception {
    LocalTime base = new LocalTime(12, 30, 45, 500);
    LocalTime result = base.plusMinutes(14400); // 24 hours
    assertEquals("12:30:45.500", result.toString());
}

@Test
public void testPlusMinutesLargeNegative() throws Exception {
    LocalTime base = new LocalTime(12, 30, 45, 500);
    LocalTime result = base.plusMinutes(-14400); // 24 hours
    assertEquals("12:30:45.500", result.toString());
}

@Test
public void testPlusMinutesMaxMinutes() throws Exception {
    LocalTime base = new LocalTime(0, 0, 0, 0);
    LocalTime result = base.plusMinutes(Integer.MAX_VALUE);
    assertEquals("00:00:00.000", result.toString());
}

@Test
public void testPlusMinutesMinMinutes() throws Exception {
    LocalTime base = new LocalTime(0, 0, 0, 0);

}
