@Test
public void testDaysBetweenSameDay() throws Exception {
    LocalDate start = new LocalDate(2023, 10, 1);
    LocalDate end = new LocalDate(2023, 10, 1);
    assertEquals(Days.days(0), Days.daysBetween(start, end));
}
