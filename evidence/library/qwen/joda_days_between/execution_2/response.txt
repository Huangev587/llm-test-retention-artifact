@Test
public void testDaysBetweenLeapYear() throws Exception {
    LocalDate startDate = new LocalDate(2020, 2, 28);
    LocalDate endDate = new LocalDate(2020, 3, 1);
    Days result = Days.daysBetween(startDate, endDate);
    assertEquals(1, result.getDays());
}
