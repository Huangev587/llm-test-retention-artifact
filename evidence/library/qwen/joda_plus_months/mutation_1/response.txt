@Test
public void testPlusMonthsEndOfMonthClamp() throws Exception {
    LocalDate date = new LocalDate(2023, 10, 31);
    LocalDate result = date.plusMonths(1);
    assertEquals("2023-11-30", result.toString());
}

@Test
public void testPlusMonthsLargePositive() throws Exception {
    LocalDate date = new LocalDate(2023, 10, 5);
    LocalDate result = date.plusMonths(1200);
    assertEquals("2100-08-05", result.toString());
}

@Test
public void testPlusMonthsLargeNegative() throws Exception {
    LocalDate date = new LocalDate(2023, 10, 5);
    LocalDate result = date.plusMonths(-1200);
    assertEquals("1900-02-05", result.toString());
}

@Test
public void testPlusMonthsCrossYear() throws Exception {
    LocalDate date = new LocalDate(2023, 12, 31);
    LocalDate result = date.plusMonths(1);
    assertEquals("2024-01-31", result.toString());
}

@Test
public void testPlusMonthsCrossYearNegative() throws Exception {
    LocalDate date = new LocalDate(2024, 1, 31);
    LocalDate result = date.plusMonths(-1);
    assertEquals("2023-12-31", result.toString());
}

@Test
public void testPlusMonthsLeapYear() throws Exception {
    LocalDate date = new LocalDate(2020, 2, 29);
    LocalDate result = date.plusMonths(1);
    assertEquals("2020-03-29", result.toString());
}

@Test
public void testPlusMonthsNonLeapYear() throws Exception {
    LocalDate date = new LocalDate(2021, 2, 28);
    LocalDate result = date.plusMonths(1);
    assertEquals("2021-03-28", result.toString());
}

@Test
public void testPlusMonthsEdgeCase() throws Exception {
    LocalDate date = new LocalDate(1900, 1, 1);
    LocalDate result = date.plusMonths(12);
    assertEquals("1901-01-01", result.toString());
}
