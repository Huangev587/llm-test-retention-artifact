@Test
public void testPlusMonthsEndOfMonthClampPositive() throws Exception {
    LocalDate date = new LocalDate(2023, 12, 31);
    LocalDate result = date.plusMonths(1);
    assertEquals("2024-01-31", result.toString());
}

@Test
public void testPlusMonthsLargePositiveDelta() throws Exception {
    LocalDate date = new LocalDate(2023, 1, 1);
    LocalDate result = date.plusMonths(1200);
    assertEquals("2123-01-01", result.toString());
}

@Test
public void testPlusMonthsLargeNegativeDelta() throws Exception {
    LocalDate date = new LocalDate(2023, 1, 1);
    LocalDate result = date.plusMonths(-1200);
    assertEquals("1923-01-01", result.toString());
}

@Test
public void testPlusMonthsFebruaryLeapYear() throws Exception {

    assertEquals("2020-03-29", result.toString());
}

public void testPlusMonthsFebruaryNonLeapYear() throws Exception {

}

public void testPlusMonthsDecemberToJanuary() throws Exception {
