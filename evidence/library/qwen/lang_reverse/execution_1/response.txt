@Test
public void testReverseLongString() throws Exception {
    String original = "abcdefghijklmnopqrstuvwxyz";
    String expected = "zyxwvutsrqponmlkjihgfedcba";
    assertEquals(expected, StringUtils.reverse(original));
}

@Test
public void testReverseMixedCase() throws Exception {
    String original = "HelloWorld";
    String expected = "dlroWolleH";
    assertEquals(expected, StringUtils.reverse(original));
}

@Test
public void testReverseWithSpaces() throws Exception {
    String original = "Hello World!";
    String expected = "!dlroW olleH";
    assertEquals(expected, StringUtils.reverse(original));
}

@Test
public void testReverseWithNumbers() throws Exception {
    String original = "1234567890";
    String expected = "0987654321";
    assertEquals(expected, StringUtils.reverse(original));
}

@Test
public void testReverseWithSpecialCharacters() throws Exception {
    String original = "!@#$%^&*()";
    String expected = ")(&*^%$#@!";
    assertEquals(expected, StringUtils.reverse(original));
}

@Test
public void testReverseWithLeadingAndTrailingSpaces() throws Exception {
    String original = "  Hello World!  ";
    String expected = "  !dlroW olleH  ";
    assertEquals(expected, StringUtils.reverse(original));
}

@Test
public void testReverseWithAllControlCharacters() throws Exception {
    String original = "\u0000\u0001\u0002\u0003\u0004\u0005\u0006\u0007\u0008\u0009\u000A\u000B\u000C\u000D\u000E\u000F" +
                      "\u0010\u0011\u0012\u0013\u0014\u0015\u0016\u0017\u0018\u0019\u001A\u001B\u001C\u001D\u001E\u001F";
    String expected = "\u001F\u001E\u001D\u001C\u001B\u001A\u0019\u0018\u0017\u0016\u0015\u0014\u0013\u0012\u0011\u0010" +
                      "\u000F\u000E\u000D\u000C\u000B\u000A\u0009\u0008\u0007\u0006\u0005\u0004\u0003\u0002\u0001\u0000";
    assertEquals(expected, StringUtils.reverse(original));
}
