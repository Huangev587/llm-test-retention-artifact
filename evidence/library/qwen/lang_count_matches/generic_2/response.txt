@Test
public void testCountMatchesNoMatch() throws Exception {

public void testCountMatchesOverlappingMatches() throws Exception {
    assertEquals(1, StringUtils.countMatches("aaa", "aa"));

public void testCountMatchesLargeTextSmallNeedle() throws Exception {
    assertEquals(10, StringUtils.countMatches("aaaaaaaaaa", "a"));
}

public void testCountMatchesSmallTextLargeNeedle() throws Exception {
    assertEquals(0, StringUtils.countMatches("abc", "abcd"));
}

@Test
public void testCountMatchesWhitespace() throws Exception {
    assertEquals(1, StringUtils.countMatches(" a ", " "));
}

@Test
public void testCountMatchesMixedCase() throws Exception {
    assertEquals(1, StringUtils.countMatches("Abc", "bc"));
}

@Test
public void testCountMatchesLeadingTrailingSpaces() throws Exception {
    assertEquals(1, StringUtils.countMatches(" ab ", "b"));
}

public void testCountMatchesSpecialCharacters() throws Exception {
    assertEquals(1, StringUtils.countMatches("!@#$$%^&*()", "!@#"));
