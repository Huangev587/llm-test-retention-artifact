    assertNull(result);
}

public void testSubarrayEmptyArray() throws Exception {
    int[] values = {};
    int[] result = subarray(values, 0, 0);
    assertNotNull(result);
    assertEquals(0, result.length);
}

public void testSubarrayFullArray() throws Exception {
    int[] values = {1, 2, 3, 4, 5};
    int[] result = subarray(values, 0, 5);
    assertNotNull(result);
    assertArrayEquals(new int[]{1, 2, 3, 4, 5}, result);
}
