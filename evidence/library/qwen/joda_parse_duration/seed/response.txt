@Test
public void testParseValidDuration() throws Exception {
    Duration duration = Duration.parse("PT1H");
    assertEquals(3600000L, duration.getMillis());
}

@Test
public void testParseValidNegativeDuration() throws Exception {
    Duration duration = Duration.parse("-PT1H");
    assertEquals(-3600000L, duration.getMillis());
}

@Test
public void testParseValidFractionalDuration() throws Exception {
    Duration duration = Duration.parse("PT1.5H");
    assertEquals(5400000L, duration.getMillis());
}

@Test
public void testParseValidLongDuration() throws Exception {
    Duration duration = Duration.parse("PT99999999999999S");
    assertEquals(99999999999999000L, duration.getMillis());
}

public void testParseValidShortDuration() throws Exception {
    Duration duration = Duration.parse("PT1S");
    assertEquals(1000L, duration.getMillis());
}

public void testParseValidZeroDuration() throws Exception {
    Duration duration = Duration.parse("PT0S");

}

public void testParseValidMaxDuration() throws Exception {
