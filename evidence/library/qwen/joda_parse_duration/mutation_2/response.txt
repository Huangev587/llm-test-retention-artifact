@Test
public void testParseValidOneSecondDuration() throws Exception {
    Duration result = Duration.parse("PT1S");
    assertEquals(1000, result.getMillis());
}

@Test
public void testParseValidNegativeOneSecondDuration() throws Exception {
    Duration result = Duration.parse("-PT1S");
    assertEquals(-1000, result.getMillis());
}
